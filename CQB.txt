-- Delta Compatible Local-Space Solid Camera (Respects Head Limits & Movement)
local Players = game:GetService("Players")
local CoreGui = game:GetService("CoreGui")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer

-- Absolute Coordinate Adjustments (X, Y, Z relative to head orientation)
local customOffset = Vector3.new(0, 0, 1.2) -- X: Left/Right, Y: Up/Down, Z: Forward (out of head)

-- Set up camera zoom limits for pinch-zoom functionality
local function setupCamera()
	player.CameraMinZoomDistance = 0.5
	player.CameraMaxZoomDistance = 20
end

player.CharacterAdded:Connect(setupCamera)
if player.Character then
	task.spawn(setupCamera)
end

-- Render loop to apply local-space head offset without blocking movement or head limits
RunService.RenderStepped:Connect(function()
	local character = player.Character
	if not character or not character:FindFirstChild("Humanoid") then return end
	
	local humanoid = character.Humanoid
	local head = character:FindFirstChild("Head")
	local camera = workspace.CurrentCamera
	
	if head then
		local camDist = (camera.CFrame.Position - head.Position).Magnitude
		
		if camDist <= 1.8 then
			-- First person active: Apply offset in local object space so it respects 
			-- the game's head rotation limits and allows normal character movement.
			humanoid.CameraOffset = customOffset
		else
			-- Zoomed out: Reset offset so default third-person camera works cleanly
			humanoid.CameraOffset = Vector3.new(0, 0, 0)
		end
	end
end)

-- Remove old GUI if it exists
if CoreGui:FindFirstChild("SolidFPCameraUI") then
	CoreGui.SolidFPCameraUI:Destroy()
end

-- Create ScreenGui
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "SolidFPCameraUI"
screenGui.ResetOnSpawn = false
screenGui.Parent = CoreGui

-- Toggle Button
local toggleBtn = Instance.new("TextButton")
toggleBtn.Size = UDim2.new(0, 110, 0, 35)
toggleBtn.Position = UDim2.new(0, 20, 0, 20)
toggleBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
toggleBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
toggleBtn.TextSize = 13
toggleBtn.Font = Enum.Font.GothamBold
toggleBtn.Text = "📷 Solid Cam"
toggleBtn.Parent = screenGui

local toggleCorner = Instance.new("UICorner")
toggleCorner.CornerRadius = UDim.new(0, 6)
toggleCorner.Parent = toggleBtn

-- Main Control Panel
local frame = Instance.new("Frame")
frame.Size = UDim2.new(0, 250, 0, 280)
frame.Position = UDim2.new(0, 20, 0, 65)
frame.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
frame.BorderSizePixel = 0
frame.Visible = false
frame.Parent = screenGui

local frameCorner = Instance.new("UICorner")
frameCorner.CornerRadius = UDim.new(0, 8)
frameCorner.Parent = frame

-- Robust Mobile-Friendly Draggable Implementation
local dragging, dragStart, startPos

frame.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = frame.Position
		
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		frame.Position = UDim2.new(
			startPos.X.Scale, 
			startPos.X.Offset + delta.X, 
			startPos.Y.Scale, 
			startPos.Y.Offset + delta.Y
		)
	end
end)

toggleBtn.MouseButton1Click:Connect(function()
	frame.Visible = not frame.Visible
	toggleBtn.BackgroundColor3 = frame.Visible and Color3.fromRGB(0, 120, 215) or Color3.fromRGB(40, 40, 40)
end)

-- Title
local title = Instance.new("TextLabel")
title.Size = UDim2.new(1, 0, 0, 30)
title.BackgroundTransparency = 1
title.Text = "Solid Coordinate Adjuster"
title.TextColor3 = Color3.fromRGB(255, 255, 255)
title.TextSize = 13
title.Font = Enum.Font.GothamBold
title.Parent = frame

-- Info Label showing exact coordinates
local infoLabel = Instance.new("TextLabel")
infoLabel.Size = UDim2.new(1, 0, 0, 25)
infoLabel.Position = UDim2.new(0, 0, 0, 30)
infoLabel.BackgroundTransparency = 1
infoLabel.TextColor3 = Color3.fromRGB(180, 180, 180)
infoLabel.TextSize = 12
infoLabel.Font = Enum.Font.GothamBold
infoLabel.Parent = frame

local function updateLabel()
	infoLabel.Text = string.format("Coords -> X: %.1f | Y: %.1f | Z: %.1f", customOffset.X, customOffset.Y, customOffset.Z)
end
updateLabel()

-- Helper function to create adjustment buttons
local function createButton(text, posY, callback)
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(0.9, 0, 0, 28)
	btn.Position = UDim2.new(0.05, 0, 0, posY)
	btn.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
	btn.TextColor3 = Color3.fromRGB(255, 255, 255)
	btn.TextSize = 11
	btn.Font = Enum.Font.Gotham
	btn.Text = text
	
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, 4)
	corner.Parent = btn
	
	btn.MouseButton1Click:Connect(function()
		callback()
		updateLabel()
	end)
	
	btn.Parent = frame
end

-- Coordinate Control Bindings
createButton("Reset to Default (0, 0, 1.2)", 60, function()
	customOffset = Vector3.new(0, 0, 1.2)
end)

createButton("X +0.5 (Right) / X -0.5 (Left)", 93, function()
	customOffset = Vector3.new(customOffset.X == 0 and 0.5 or (customOffset.X == 0.5 and -0.5 or 0), customOffset.Y, customOffset.Z)
end)

createButton("Y +0.5 (Up) / Y -0.5 (Down)", 126, function()
	customOffset = Vector3.new(customOffset.X, customOffset.Y == 0.5 and 1.0 or (customOffset.Y == 1.0 and 0 or 0.5), customOffset.Z)
end)

createButton("Z +0.5 (Forward - Out of head)", 159, function()
	customOffset = customOffset + Vector3.new(0, 0, 0.5)
end)

createButton("Z -0.5 (Backward - Into head)", 192, function()
	customOffset = customOffset - Vector3.new(0, 0, 0.5)
end)

createButton("Close Menu", 235, function()
	frame.Visible = false
	toggleBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
end)

print("Compatible He-Limit Solid Camera loaded successfully!")
