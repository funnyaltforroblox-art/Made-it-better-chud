-- Delta Compatible First-Person Script with Pinch-Zoom Toggle & Draggable UI
local Players = game:GetService("Players")
local CoreGui = game:GetService("CoreGui")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer
local camera = workspace.CurrentCamera

local isFirstPerson = true

-- Function to handle camera zoom and first-person enforcement
local function setupFirstPerson()
	local character = player.Character or player.CharacterAdded:Wait()
	local humanoid = character:WaitForChild("Humanoid")
	
	-- Allow normal zoom bounds so pinch zoom works
	player.CameraMinZoomDistance = 0
	player.CameraMaxZoomDistance = 20 -- Allow zooming out up to 20 studs
end

player.CharacterAdded:Connect(setupFirstPerson)
if player.Character then
	task.spawn(setupFirstPerson)
end

-- Monitor camera distance to automatically toggle first person state via pinch zoom
RunService.RenderStepped:Connect(function()
	local character = player.Character
	if not character or not character:FindFirstChild("Head") then return end
	
	-- Calculate distance from camera to head to detect if user pinch-zoomed out
	local headPos = character.Head.Position
	local camDist = (camera.CFrame.Position - headPos).Magnitude
	
	-- If zoom distance is very small, lock to true first person
	if camDist <= 1.0 then
		isFirstPerson = true
		player.CameraMinZoomDistance = 0
	else
		-- User has pinch-zoomed out
		isFirstPerson = false
	end
end)

-- Remove old GUI if it exists
if CoreGui:FindFirstChild("AdvancedFPCameraUI") then
	CoreGui.AdvancedFPCameraUI:Destroy()
end

-- Create ScreenGui
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "AdvancedFPCameraUI"
screenGui.ResetOnSpawn = false
screenGui.Parent = CoreGui

-- Toggle Button (To open/close the menu)
local toggleBtn = Instance.new("TextButton")
toggleBtn.Size = UDim2.new(0, 110, 0, 35)
toggleBtn.Position = UDim2.new(0, 20, 0, 20)
toggleBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
toggleBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
toggleBtn.TextSize = 13
toggleBtn.Font = Enum.Font.GothamBold
toggleBtn.Text = "📷 FP Menu"
toggleBtn.Parent = screenGui

local toggleCorner = Instance.new("UICorner")
toggleCorner.CornerRadius = UDim.new(0, 6)
toggleCorner.Parent = toggleBtn

-- Main Draggable Control Panel
local frame = Instance.new("Frame")
frame.Size = UDim2.new(0, 240, 0, 260)
frame.Position = UDim2.new(0, 20, 0, 65)
frame.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
frame.BorderSizePixel = 0
frame.Visible = false -- Starts hidden, controlled by toggle button
frame.Active = true
frame.Draggable = true
frame.Parent = screenGui

local frameCorner = Instance.new("UICorner")
frameCorner.CornerRadius = UDim.new(0, 8)
frameCorner.Parent = frame

-- Toggle functionality
toggleBtn.MouseButton1Click:Connect(function()
	frame.Visible = not frame.Visible
	toggleBtn.BackgroundColor3 = frame.Visible and Color3.fromRGB(0, 120, 215) or Color3.fromRGB(40, 40, 40)
end)

-- Title
local title = Instance.new("TextLabel")
title.Size = UDim2.new(1, 0, 0, 30)
title.BackgroundTransparency = 1
title.Text = "Head Camera Adjuster"
title.TextColor3 = Color3.fromRGB(255, 255, 255)
title.TextSize = 13
title.Font = Enum.Font.GothamBold
title.Parent = frame

-- Status / Values Display
local infoLabel = Instance.new("TextLabel")
infoLabel.Size = UDim2.new(1, 0, 0, 25)
infoLabel.Position = UDim2.new(0, 0, 0, 30)
infoLabel.BackgroundTransparency = 1
infoLabel.TextColor3 = Color3.fromRGB(180, 180, 180)
infoLabel.TextSize = 11
infoLabel.Font = Enum.Font.Gotham
infoLabel.Text = "Offset: X:0 | Y:0 | Z:0"
infoLabel.Parent = frame

-- Function to update offset display text
local function updateInfoText()
	local char = player.Character
	if char and char:FindFirstChild("Humanoid") then
		local o = char.Humanoid.CameraOffset
		infoLabel.Text = string.format("Offset: X:%.1f | Y:%.1f | Z:%.1f", o.X, o.Y, o.Z)
	end
end

-- Helper to create adjustment buttons
local function createButton(text, posY, callback)
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(0.9, 0, 0, 26)
	btn.Position = UDim2.new(0.05, 0, 0, posY)
	btn.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
	btn.TextColor3 = Color3.fromRGB(255, 255, 255)
	btn.TextSize = 11
	btn.Font = Enum.Font.Gotham
	btn.Text = text
	
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, 4)
	corner.Parent = btn
	
	btn.MouseButton1Click:Connect(function()
		callback()
		updateInfoText()
	end)
	
	btn.Parent = frame
end

-- Button Bindings for X, Y, Z Adjustments
createButton("Reset Offset (0, 0, 0)", 60, function()
	local char = player.Character
	if char and char:FindFirstChild("Humanoid") then
		char.Humanoid.CameraOffset = Vector3.new(0, 0, 0)
	end
end)

createButton("Move Up (+0.5 Y) / Down (-0.5 Y)", 91, function()
	local char = player.Character
	if char and char:FindFirstChild("Humanoid") then
		local cur = char.Humanoid.CameraOffset
		char.Humanoid.CameraOffset = Vector3.new(cur.X, (cur.Y == 0 and 0.5 or (cur.Y == 0.5 and -0.5 or 0)), cur.Z)
	end
end)

createButton("Push Forward (+0.5 Z)", 122, function()
	local char = player.Character
	if char and char:FindFirstChild("Humanoid") then
		char.Humanoid.CameraOffset += Vector3.new(0, 0, 0.5)
	end
end)

createButton("Pull Backward (-0.5 Z)", 153, function()
	local char = player.Character
	if char and char:FindFirstChild("Humanoid") then
		char.Humanoid.CameraOffset -= Vector3.new(0, 0, 0.5)
	end
end)

createButton("Shift Left (-0.5 X) / Right (+0.5 X)", 184, function()
	local char = player.Character
	if char and char:FindFirstChild("Humanoid") then
		local cur = char.Humanoid.CameraOffset
		char.Humanoid.CameraOffset = Vector3.new((cur.X == 0 and -0.5 or (cur.X == -0.5 and 0.5 or 0)), cur.Y, cur.Z)
	end
end)

createButton("Close Menu", 220, function()
	frame.Visible = false
	toggleBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
end)

print("Dynamic Pinch-Zoom First-Person Camera loaded!")
