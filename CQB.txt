-- Delta Compatible Solid Camera (Tracks Torso/Head Rotations & Custom Sensitivity)
local Players = game:GetService("Players")
local CoreGui = game:GetService("CoreGui")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer
local camera = workspace.CurrentCamera

-- Absolute Coordinate Adjustments (X: Left/Right, Y: Up/Down, Z: Forward/Back)
local customOffset = Vector3.new(0, 0.5, 1.2)

-- Sensitivity multiplier to fine-tune look speed
local lookSensitivity = 1.5

-- Manual view look offsets added on top of head rotation
local yawOffset = 0
local pitchOffset = 0

-- Tracking for touch panning (ignoring left-side movement thumbstick)
local panningInput = nil
local lastPanPosition = nil

-- Setup zoom limits for pinch-zoom out
local function setupCamera()
	pcall(function()
		player.CameraMinZoomDistance = 0.5
		player.CameraMaxZoomDistance = 20
	end)
end

player.CharacterAdded:Connect(setupCamera)
if player.Character then
	task.spawn(setupCamera)
end

-- Isolate touch panning to the right side of the screen
UserInputService.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton2 then
		local pos = input.Position
		-- Ignore left 35% of the screen where movement joysticks usually reside
		if pos.X > camera.ViewportSize.X * 0.35 then
			panningInput = input
			lastPanPosition = pos
		end
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input == panningInput then
		panningInput = nil
		lastPanPosition = nil
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if input == panningInput then
		local currentPos = input.Position
		local delta = currentPos - lastPanPosition
		lastPanPosition = currentPos
		
		local character = player.Character
		if character and character:FindFirstChild("Head") then
			local camDist = (camera.CFrame.Position - character.Head.Position).Magnitude
			if camDist <= 2.0 then
				yawOffset = yawOffset - delta.X * 0.003 * lookSensitivity
				pitchOffset = math.clamp(pitchOffset - delta.Y * 0.003 * lookSensitivity, -1.5, 1.5)
			end
		end
	end
end)

-- Main camera update loop tracking head/torso rotation matrices
RunService.RenderStepped:Connect(function()
	pcall(function()
		local character = player.Character
		if not character then return end
		
		local head = character:FindFirstChild("Head")
		if not head then return end
		
		local camDist = (camera.CFrame.Position - head.Position).Magnitude
		
		if camDist <= 2.0 then
			camera.CameraType = Enum.CameraType.Scriptable
			
			-- Extract head's actual rotation (which is driven by torso twists and head-look mechanics)
			local headRotationOnly = head.CFrame - head.Position
			
			-- Combine head rotation with user look offset angles
			local adjustedRotation = headRotationOnly * CFrame.Angles(0, yawOffset, 0) * CFrame.Angles(pitchOffset, 0, 0)
			
			-- Position camera rigidly at head position using exact X, Y, Z coordinates
			camera.CFrame = CFrame.new(head.Position) * adjustedRotation * CFrame.new(customOffset)
		else
			camera.CameraType = Enum.CameraType.Custom
			yawOffset = 0
			pitchOffset = 0
		end
	end)
end)

-- Remove old GUI if it exists
if CoreGui:FindFirstChild("SolidFPCameraUI") then
	CoreGui.SolidFPCameraUI:Destroy()
end

-- Create ScreenGui securely in CoreGui
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "SolidFPCameraUI"
screenGui.ResetOnSpawn = false
screenGui.Parent = CoreGui

-- Toggle Button
local toggleBtn = Instance.new("TextButton")
toggleBtn.Size = UDim2.new(0, 110, 0, 35)
toggleBtn.Position = UDim2.new(0, 20, 0, 20)
toggleBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
toggleBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
toggleBtn.TextSize = 13
toggleBtn.Font = Enum.Font.GothamBold
toggleBtn.Text = "📷 Solid Cam"
toggleBtn.Parent = screenGui

local toggleCorner = Instance.new("UICorner")
toggleCorner.CornerRadius = UDim.new(0, 6)
toggleCorner.Parent = toggleBtn

-- Main Control Panel Frame
local frame = Instance.new("Frame")
frame.Size = UDim2.new(0, 250, 0, 310)
frame.Position = UDim2.new(0, 20, 0, 65)
frame.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
frame.BorderSizePixel = 0
frame.Visible = false
frame.Parent = screenGui

local frameCorner = Instance.new("UICorner")
frameCorner.CornerRadius = UDim.new(0, 8)
frameCorner.Parent = frame

-- Mobile-Friendly Draggable Logic
local dragging, dragStart, startPos

frame.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = frame.Position
		
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		frame.Position = UDim2.new(
			startPos.X.Scale, 
			startPos.X.Offset + delta.X, 
			startPos.Y.Scale, 
			startPos.Y.Offset + delta.Y
		)
	end
end)

toggleBtn.MouseButton1Click:Connect(function()
	frame.Visible = not frame.Visible
	toggleBtn.BackgroundColor3 = frame.Visible and Color3.fromRGB(0, 120, 215) or Color3.fromRGB(40, 40, 40)
end)

-- Title
local title = Instance.new("TextLabel")
title.Size = UDim2.new(1, 0, 0, 25)
title.BackgroundTransparency = 1
title.Text = "Solid Cam Adjuster"
title.TextColor3 = Color3.fromRGB(255, 255, 255)
title.TextSize = 13
title.Font = Enum.Font.GothamBold
title.Parent = frame

-- Info Label showing coordinates and sensitivity
local infoLabel = Instance.new("TextLabel")
infoLabel.Size = UDim2.new(1, 0, 0, 25)
infoLabel.Position = UDim2.new(0, 0, 0, 25)
infoLabel.BackgroundTransparency = 1
infoLabel.TextColor3 = Color3.fromRGB(180, 180, 180)
infoLabel.TextSize = 11
infoLabel.Font = Enum.Font.GothamBold
infoLabel.Parent = frame

local function updateLabel()
	infoLabel.Text = string.format("XYZ: (%.1f, %.1f, %.1f) | Sens: %.1f", customOffset.X, customOffset.Y, customOffset.Z, lookSensitivity)
end
updateLabel()

-- Helper function to create adjustment buttons
local function createButton(text, posY, callback)
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(0.9, 0, 0, 25)
	btn.Position = UDim2.new(0.05, 0, 0, posY)
	btn.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
	btn.TextColor3 = Color3.fromRGB(255, 255, 255)
	btn.TextSize = 11
	btn.Font = Enum.Font.Gotham
	btn.Text = text
	
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, 4)
	corner.Parent = btn
	
	btn.MouseButton1Click:Connect(function()
		callback()
		updateLabel()
	end)
	
	btn.Parent = frame
end

-- Control Bindings
createButton("Reset Defaults", 52, function()
	customOffset = Vector3.new(0, 0.5, 1.2)
	lookSensitivity = 1.5
	yawOffset = 0
	pitchOffset = 0
end)

createButton("Sensitivity +0.5 / -0.5 (Cycle)", 80, function()
	lookSensitivity = lookSensitivity + 0.5
	if lookSensitivity > 3.5 then lookSensitivity = 1.0 end
end)

createButton("X +0.5 (Right) / X -0.5 (Left)", 108, function()
	customOffset = Vector3.new(customOffset.X == 0 and 0.5 or (customOffset.X == 0.5 and -0.5 or 0), customOffset.Y, customOffset.Z)
end)

createButton("Y +0.5 (Up) / Y -0.5 (Down)", 136, function()
	customOffset = Vector3.new(customOffset.X, customOffset.Y == 0.5 and 1.0 or (customOffset.Y == 1.0 and 0 or 0.5), customOffset.Z)
end)

createButton("Z +0.5 (Forward) / Z -0.5 (Back)", 164, function()
	customOffset = customOffset + Vector3.new(0, 0, 0.5)
	if customOffset.Z > 2.5 then customOffset = Vector3.new(customOffset.X, customOffset.Y, 0.5) end
end)

createButton("Quick Center Head Offset", 192, function()
	customOffset = Vector3.new(0, 0, 0)
end)

createButton("Close Menu", 235, function()
	frame.Visible = false
	toggleBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
end)

print("Torso/Head Tracking Solid Camera loaded!")
