-- Dragging Logic (Mouse + Touch)
local dragging, dragInput, dragStart, startPos
mainFrame.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = mainFrame.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

mainFrame.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		dragInput = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if input == dragInput and dragging then
		local delta = input.Position - dragStart
		mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
end)

-- Health Application Logic
local function applyClientHP()
	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return end

	local rawText = hpBox.Text:gsub("%s+", "")
	if rawText == "" then return end

	local newHealth
	if rawText:sub(-1) == "%" then
		local pct = tonumber(rawText:sub(1, -2))
		if pct then
			newHealth = (math.clamp(pct, 0, 100) / 100) * humanoid.MaxHealth
		end
	else
		local val = tonumber(rawText)
		if val then
			newHealth = math.clamp(val, 0, humanoid.MaxHealth)
		end
	end

	if newHealth then
		-- Prevent accidental client-side death state if testing 0 HP
		if newHealth <= 0 then
			humanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
		end
		targetHP = newHealth
		humanoid.Health = newHealth
	end
end

applyBtn.MouseButton1Click:Connect(applyClientHP)
hpBox.FocusLost:Connect(function(enterPressed)
	if enterPressed then
		applyClientHP()
	end
end)

lockBtn.MouseButton1Click:Connect(function()
	isLocked = not isLocked
	if isLocked then
		lockBtn.Text = "LOCK HP: ON"
		lockBtn.BackgroundColor3 = Color3.fromRGB(35, 95, 55)
		lockBtn.TextColor3 = Color3.fromRGB(120, 255, 160)
		if not targetHP then applyClientHP() end
	else
		lockBtn.Text = "LOCK HP: OFF"
		lockBtn.BackgroundColor3 = Color3.fromRGB(60, 35, 35)
		lockBtn.TextColor3 = Color3.fromRGB(240, 120, 120)
	end
end)

-- Live Display & HP Lock Enforcement
RunService.RenderStepped:Connect(function()
	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if humanoid and humanoid.MaxHealth > 0 then
		if isLocked and targetHP ~= nil and math.abs(humanoid.Health - targetHP) > 0.01 then
			humanoid.Health = targetHP
		end
		local pct = math.floor((humanoid.Health / humanoid.MaxHealth) * 100 + 0.5)
		statusLabel.Text = string.format("HP: %.0f / %.0f (%d%%)", humanoid.Health, humanoid.MaxHealth, pct)
	else
		statusLabel.Text = "HP: -- / -- (--%)"
	end
end)
